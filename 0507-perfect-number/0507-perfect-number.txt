    public boolean checkPerfectNumber(int num) {
        if (num <= 1) return false;
        int count=1;

        for(int i=2;i<=num/2;i++){
           if(num%i==0){
            count+=i;}
           
    }
     if(count==num){
             return true;}
    return false; 
}
